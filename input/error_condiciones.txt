    entero x;
    cadena texto;
    caracter letra;

    texto = "hola";
    letra = 'a';
    
    // Si con condición no booleana/entera
    si (texto) {  // ERROR: Condición debe ser booleana o entera
        x = 5;
    }
    
    si (letra) {  // ERROR: Condición debe ser booleana o entera
        x = 10;
    }
    
    // Mientras con condición incorrecta
    mientras (texto) {  // ERROR: Condición debe ser booleana o entera
        x = x + 1;
    }
    
    mientras (letra) {  // ERROR: Condición debe ser booleana o entera
        x = x - 1;
    }
    
    // Para con condición incorrecta
    para (x = 0; texto; x = x + 1) {  // ERROR: Condición debe ser booleana o entera
        escribir(x);

    para (x = 0; letra; x = x + 1) {  // ERROR: Condición debe ser booleana o entera
        escribir(x);
    }
    
    // Condiciones correctas para comparar
    si (x > 0) {        // OK: comparación devuelve booleano
        escribir(x);

        escribir(x);
